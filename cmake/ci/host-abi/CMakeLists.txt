cmake_minimum_required(VERSION 3.20)
project(TurboScriptInstalledHostABI LANGUAGES C CXX)
include(CTest)
find_package(TurboScript CONFIG REQUIRED)
foreach(language IN ITEMS c cpp)
  add_executable(host_abi_${language} host_contract.${language})
  target_link_libraries(host_abi_${language} PRIVATE TurboScript::TurboScript)
  set_target_properties(host_abi_${language} PROPERTIES
    C_STANDARD 11 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF
    CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
  if(MSVC)
    target_compile_options(host_abi_${language} PRIVATE /W4 /WX)
  else()
    target_compile_options(host_abi_${language} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
  foreach(mode IN ITEMS interpreter jit)
    add_test(NAME installed_host_${language}_${mode} COMMAND host_abi_${language} ${mode})
  endforeach()
endforeach()
